# 基础设施
如果不是放在同一个compose文件下或者没有手动创建共享网络，别的服务需要使用基础设施需要设置：
```yaml
extra_hosts:
      - 'host.docker.internal:host-gateway' 
```yaml
然后通过host.docker.internal:端口进行连接

services:
  # MySQL 8.0 服务
  mysql:
    image: mysql:8.0
    container_name: mysql
    restart: always
    ports:
      - "3306:3306"  # 主机端口:容器端口
    environment:
      MYSQL_ROOT_PASSWORD: 123456.  # root 密码（务必修改）
      MYSQL_DATABASE: default_db       # 自动创建的默认数据库
      TZ: Asia/Shanghai
    volumes:
      - ./mysql/data:/var/lib/mysql    # 数据持久化
      - ./mysql/conf:/etc/mysql/conf.d
    networks:
      - app-network

  # Redis 服务（带密码、持久化）
  redis:
    image: redis:7
    container_name: redis
    restart: always
    ports:
      - "6379:6379"
    command: redis-server --requirepass 123456 --appendonly yes  # 密码 + 持久化
    volumes:
      - ./redis/data:/data
    networks:
      - app-network

# 自定义网络，让两个容器互相通信
networks:
  app-network:
    driver: bridge
```

# 监控
## docker compose
```yaml
services:
  # Prometheus服务
  prometheus:
    image: prom/prometheus:latest
    container_name: prometheus
    restart: unless-stopped
    ports:
      - "9090:9090"  # 对外暴露9090端口
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml  # 挂载配置文件
      - ./prometheus_data:/prometheus  # 持久化数据
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
    user: root

  # Node Exporter（收集系统CPU、内存、磁盘等指标）
  node_exporter:
    image: prom/node-exporter:latest
    container_name: node_exporter
    restart: unless-stopped
    ports:
      - "9100:9100"
    volumes:
      - /proc:/host/proc:ro
      - /sys:/host/sys:ro
      - /:/rootfs:ro
    command:
      - '--path.procfs=/host/proc'
      - '--path.sysfs=/host/sys'
      - '--path.rootfs=/rootfs'  # 显式指定根文件系统路径
      - '--collector.filesystem.ignored-mount-points=^/(sys|proc|dev|host|etc)($$|/)'

  # Grafana服务
  grafana:
    image: grafana/grafana:latest
    container_name: grafana
    restart: unless-stopped
    ports:
      - "3000:3000"  # 对外暴露3000端口
    volumes:
      - ./grafana_data:/var/lib/grafana  # 持久化Grafana数据
    user: root  # 指定以root用户运行容器
    environment:
      - GF_SECURITY_ADMIN_USER=admin  # Grafana默认管理员账号
      - GF_SECURITY_ADMIN_PASSWORD=admin  # 默认密码（建议登录后修改）
      - GF_USERS_ALLOW_SIGN_UP=false
    depends_on:
      - prometheus  # 依赖Prometheus，确保先启动Prometheus
```

## frpc
### frpc.service
```shell
[Unit]
Description=Frp Client Service
After=network.target

[Service]
Type=simple
User=root
ExecStart=/opt/frp/frpc -c /opt/frp/frpc.toml
Restart=on-failure

[Install]
WantedBy=multi-user.target
```
### frpc.toml
```yaml
# frpc.toml - 客户端基础配置
serverAddr = "1.1.1.1"   # 替换为服务端公网 IP
serverPort = 1234        # 与服务端 bindPort 一致
auth.token = "123456"  # 与服务端一致


[[proxies]]
name = "mysql"
type = "tcp"
localIP = "127.0.0.1"
localPort = 3306
remotePort = 3306

[[proxies]]
name = "redis"
type = "tcp"
localIP = "127.0.0.1"
localPort = 6379
remotePort = 6379
```
## frps
### frps.service
```yaml
[Unit]
Description=Frp Server Service
After=network.target

[Service]
Type=simple
User=root
ExecStart=/opt/frp/frps -c /opt/frp/frps.toml
Restart=on-failure

[Install]
WantedBy=multi-user.target
```
### frps.toml
```yaml
# frps.toml - 服务端基础配置
bindPort = 1234          # 客户端连接端口（必须开放防火墙）
# bindUDPPort = 1235     # xtcp 打洞 UDP 端口（可选）

# 身份验证（必须设置，客户端需一致）
auth.token = "123456"

# Web 仪表盘（可选，用于监控）
webServer.addr = "0.0.0.0"
webServer.port = 1236    # 仪表盘端口
webServer.user = "admin"
webServer.password = "123456"

# 日志配置
log.level = "info"
log.to = "/var/log/frps.log"
log.maxDays = 7
```
